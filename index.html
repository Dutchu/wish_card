package com.universe.greetings

data class Person(val name: String, val age: Int)

class BirthdayWish(private val forPerson: Person) {
    private val messages = mutableListOf<String>()
    private var projectedLongevity = 100

    fun wishAge(age: Int) = apply { projectedLongevity = age }
    fun addLine(message: String) = apply { messages.add(message) }

    fun deliver() {
        println("To: ${forPerson.name} (Level: ${forPerson.age})")
        println("Subject: Happy Birthday! Patch Notes:")
        println("--------------------------------------")
        println("  + Wishing you at least $projectedLongevity years of uptime.")
        messages.forEach { println("  + $it") }
    }
}

fun main() {
    val birthdayPerson = Person(name = "Gucek", age = 30)

    // Building the wish object with method chaining
    val wish = BirthdayWish(birthdayPerson)
        .wishAge(120)
        .addLine("May your bugs be few.")
        .addLine("And your features be many.")

    // Delivering the fully constructed wish
    wish.deliver()
}
